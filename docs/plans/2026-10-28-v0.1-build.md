```python
"""project-due-diligence 共享库：API 抓取/退避、归一化、JSON IO、运行区工具。stdlib only。"""
import json, os, re, tempfile, time, unicodedata, urllib.error, urllib.parse, urllib.request

UNIFIED_FIELDS = ["source", "source_id", "doi", "title", "year", "authors", "venue",
                  "abstract", "citation_count", "oa_url", "paper_type"]
PATHS = ["综述", "奠基", "前沿", "引文追踪", "反证"]
TIERS = {
    "快速": {"core_questions": 3, "min_included": 10, "min_reviews": 1, "l2_min": 5, "l3_min": 0, "min_gaps": 1},
    "标准": {"core_questions": 5, "min_included": 25, "min_reviews": 3, "l2_min": 10, "l3_min": 2, "min_gaps": 3},
    "深入": {"core_questions": 8, "min_included": 40, "min_reviews": 5, "l2_min": 20, "l3_min": 5, "min_gaps": 5},
}
REPORT_SECTIONS = ["Executive Summary", "项目分解", "文献检索协议（含覆盖限制）", "研究现状",
                   "痛点分析", "研究缺口台账", "创新性审计", "候选研究方向",
                   "可行性与风险", "推荐路线图", "证据附录"]

class LibError(RuntimeError):
    pass

def today() -> str:
    return time.strftime("%Y-%m-%d")

def norm_doi(doi: str) -> str:
    if not doi:
        return ""
    d = str(doi).strip().lower()
    for pre in ("https://doi.org/", "http://doi.org/", "doi.org/", "doi:"):
        if d.startswith(pre):
            d = d[len(pre):]
    return d.strip()

def norm_title(title: str) -> str:
    if not title:
        return ""
    t = unicodedata.normalize("NFKD", str(title))
    t = "".join(c for c in t if not unicodedata.combining(c))
    return re.sub(r"[^a-z0-9]+", "", t.lower())

class RateLimiter:
    def __init__(self, min_interval: float):
        self.min_interval = min_interval
        self._last = 0.0
    def wait(self):
        now = time.time()
        if self._last and now - self._last < self.min_interval:
            time.sleep(self.min_interval - (now - self._last))
        self._last = time.time()

def fetch_json(url, params=None, headers=None, retries=4, timeout=30, limiter=None):
    if params:
        url = url + ("&" if "?" in url else "?") + urllib.parse.urlencode(params)
    last = None
    for attempt in range(retries):
        if limiter:
            limiter.wait()
        req = urllib.request.Request(
            url, headers={"User-Agent": "project-due-diligence/0.1",
                          **(headers or {})})
        try:
            with urllib.request.urlopen(req, timeout=timeout) as r:
                return json.loads(r.read().decode("utf-8"))
        except urllib.error.HTTPError as e:
            if e.code in (429, 500, 502, 503, 504):
                last = e
                time.sleep(min(2 ** attempt * 2, 30))
                continue
            raise LibError(f"HTTP {e.code} {url}")
        except (urllib.error.URLError, TimeoutError, json.JSONDecodeError, OSError) as e:
            last = e
            time.sleep(min(2 ** attempt * 2, 30))
    raise LibError(f"fetch failed after {retries} tries: {url} ({last})")

def save_json(path, data):
    ap = os.path.abspath(path)
    os.makedirs(os.path.dirname(ap) or ".", exist_ok=True)
    fd, tmp = tempfile.mkstemp(dir=os.path.dirname(ap), suffix=".tmp")
    with os.fdopen(fd, "w", encoding="utf-8") as f:
        json.dump(data, f, ensure_ascii=False, indent=1)
    os.replace(tmp, path)

def load_json(path):
    if not os.path.exists(path):
        return None
    with open(path, encoding="utf-8") as f:
        return json.load(f)

def raw_path(run_dir, source):
    os.makedirs(os.path.join(run_dir, "raw"), exist_ok=True)
    return os.path.join(run_dir, "raw", f"{source}_{time.strftime('%Y%m%dT%H%M%S')}_{os.getpid()}.json")

def append_search_log(run_dir, entry):
    p = os.path.join(run_dir, "search_log.json")
    log = load_json(p) or []
    entry = dict(entry)
    entry["id"] = f"S{len(log) + 1:03d}"
    entry.setdefault("ts", time.strftime("%Y-%m-%dT%H:%M:%S"))
    log.append(entry)
    save_json(p, log)
    return entry["id"]

def load_records(run_dir):
    rawdir = os.path.join(run_dir, "raw")
    out = []
    if not os.path.isdir(rawdir):
        return out
    for fn in sorted(os.listdir(rawdir)):
        if fn.endswith(".json"):
            out.extend(load_json(os.path.join(rawdir, fn)) or [])
    return out

def default_entry(rec):
    access = "无法获取"
    if rec.get("oa_url"):
        access = "全文可读"
    elif rec.get("abstract"):
        access = "仅摘要"
    return {"paper_id": rec.get("paper_id", ""), "source": rec.get("source", ""),
            "source_id": rec.get("source_id", ""), "doi": rec.get("doi", ""),
            "title": rec.get("title", ""), "year": rec.get("year"),
            "authors": rec.get("authors", []), "venue": rec.get("venue", ""),
            "abstract": rec.get("abstract", ""), "citation_count": rec.get("citation_count", 0) or 0,
            "oa_url": rec.get("oa_url", ""), "paper_type": rec.get("paper_type", "other"),
            "n_sources": rec.get("n_sources", 1),
            "question_ids": [], "access_status": access, "reading_status": "待筛",
            "include_reason": "", "exclude_reason": "", "reading_level": "L1",
            "first_seen": today()}
```

```python
"""OpenAlex 检索（主发现引擎）。stdlib only。"""
import argparse, sys
import lib

API = "https://api.openalex.org/works"
TYPE_MAP = {"review": "review", "article": "article", "preprint": "preprint"}
LIMITER = lib.RateLimiter(0.15)

def abstract_from_inverted(inv):
    if not inv:
        return ""
    pos = {}
    for word, idxs in (inv or {}).items():
        for i in idxs:
            pos[i] = word
    return " ".join(pos[i] for i in sorted(pos))

def normalize(w):
    loc = w.get("primary_location") or {}
    src = loc.get("source") or {}
    return {"source": "openalex",
            "source_id": (w.get("id") or "").rsplit("/", 1)[-1],
            "doi": lib.norm_doi(w.get("doi") or ""),
            "title": w.get("title") or "",
            "year": w.get("publication_year"),
            "authors": [a.get("author", {}).get("display_name", "")
                        for a in (w.get("authorships") or [])[:10]],
            "venue": src.get("display_name") or "",
            "abstract": abstract_from_inverted(w.get("abstract_inverted_index")),
            "citation_count": w.get("cited_by_count", 0) or 0,
            "oa_url": loc.get("pdf_url") or (w.get("best_oa_location") or {}).get("pdf_url") or "",
            "paper_type": TYPE_MAP.get(w.get("type"), "other")}

def search(query, year_from=None, year_to=None, ptype=None, limit=50, mailto=None):
    filt = []
    if year_from:
        filt.append(f"from_publication_date:{year_from}-01-01")
    if year_to:
```

```python
"""Semantic Scholar 检索与引文双向扩展。stdlib only。S2_API_KEY 环境变量可选。"""
import argparse, os, sys
import lib

BASE = "https://api.semanticscholar.org/graph/v1"
FIELDS = "title,abstract,year,authors,venue,citationCount,externalIds,publicationTypes,openAccessPdf"
LIMITER = lib.RateLimiter(1.2)

def _headers():
    key = os.environ.get("S2_API_KEY", "")
    return {"x-api-key": key} if key else {}

def normalize(p):
    kinds = p.get("publicationTypes") or []
    return {"source": "s2", "source_id": p.get("paperId") or "",
            "doi": lib.norm_doi((p.get("externalIds") or {}).get("DOI") or ""),
            "title": p.get("title") or "", "year": p.get("year"),
            "authors": [a.get("name", "") for a in (p.get("authors") or [])[:10]],
            "venue": p.get("venue") or "", "abstract": p.get("abstract") or "",
            "citation_count": p.get("citationCount", 0) or 0,
            "oa_url": (p.get("openAccessPdf") or {}).get("url") or "",
            "paper_type": "review" if any("review" in str(k).lower() for k in kinds) else "article"}

def search(query, limit=100):
    out, offset = [], 0
    while len(out) < limit:
        data = lib.fetch_json(f"{BASE}/paper/search",
                              params={"query": query, "limit": min(100, limit - len(out)),
                                      "offset": offset, "fields": FIELDS},
                              headers=_headers(), limiter=LIMITER)
        batch = data.get("data") or []
        out.extend(batch)
        if data.get("next") is None or not batch:
            break
        offset = data["next"]
    return out[:limit]

def expand(paper_id, mode, limit=200):
    """mode: 'refs'（它引了谁）或 'cites'（谁引了它）。返回内层论文列表。"""
    ep = "references" if mode == "refs" else "citations"
    inner = "citedPaper" if mode == "refs" else "citingPaper"
    out, offset = [], 0
    while len(out) < limit:
        data = lib.fetch_json(f"{BASE}/paper/{paper_id}/{ep}",
                              params={"limit": min(1000, limit - len(out)), "offset": offset,
                                      "fields": FIELDS},
                              headers=_headers(), limiter=LIMITER)
        batch = [row.get(inner) for row in (data.get("data") or []) if row.get(inner)]
        out.extend(batch)
        if data.get("next") is None or not batch:
            break
        offset = data["next"]
    return out[:limit]

def main(argv=None):
    ap = argparse.ArgumentParser(description="Semantic Scholar search / citation expansion")
    ap.add_argument("--run", required=True)
    ap.add_argument("--query")
    ap.add_argument("--question", default="")
    ap.add_argument("--path", required=True, choices=lib.PATHS)
    ap.add_argument("--purpose", default="")
    ap.add_argument("--limit", type=int, default=100)
    ap.add_argument("--expand-refs")
    ap.add_argument("--expand-cites")
    a = ap.parse_args(argv)
    mode, pid = ("refs", a.expand_refs) if a.expand_refs else (
        ("cites", a.expand_cites) if a.expand_cites else (None, None))
    if mode:
        papers = expand(pid, mode, a.limit)
        q = f"expand:{mode}:{pid}"
    elif a.query:
        papers = search(a.query, a.limit)
        q = a.query
    else:
        ap.error("需要 --query 或 --expand-refs/--expand-cites")
    recs = [normalize(p) for p in papers]
    p = lib.raw_path(a.run, "s2")
    lib.save_json(p, recs)
    sid = lib.append_search_log(a.run, {
        "source": "s2", "query": q, "path": a.path, "question_id": a.question,
        "purpose": a.purpose, "filters": {"limit": a.limit}, "result_count": len(recs)})
    print(f"[s2] {len(recs)} records -> {p} (log {sid})")
    return 0

if __name__ == "__main__":
    sys.exit(main())
```

```python
import argparse, re, sys
```

```python
LIMITER = lib.RateLimiter(3.0)

def compose_query(q):
    return " AND ".join(f"all:{t}" for t in q.split())

def normalize(e):
    def txt(tag):
        el = e.find(f"a:{tag}", NS)
        return (el.text or "").strip() if el is not None and el.text else ""
    year = None
    if txt("published"):
        year = int(txt("published")[:4])

        if link.get("title") == "pdf":

                        if a.find("a:name", NS) is not None][:10],

            "citation_count": 0, "oa_url": pdf, "paper_type": "preprint"}
```

```python
def fetch_text(url, params=None, headers=None, retries=4, timeout=30, limiter=None):
    """GET 返回原始文本（用于 XML API）。退避策略同 fetch_json。"""
    if params:
        url = url + ("&" if "?" in url else "?") + urllib.parse.urlencode(params)
    last = None
    for attempt in range(retries):
        if limiter:
            limiter.wait()
        req = urllib.request.Request(
            url, headers={"User-Agent": "project-due-diligence/0.1", **(headers or {})})
        try:
            with urllib.request.urlopen(req, timeout=timeout) as r:
                return r.read().decode("utf-8")
        except urllib.error.HTTPError as e:
            if e.code in (429, 500, 502, 503, 504):
                last = e
                time.sleep(min(2 ** attempt * 2, 30))
                continue
            raise LibError(f"HTTP {e.code} {url}")
        except (urllib.error.URLError, TimeoutError, OSError) as e:
            last = e
            time.sleep(min(2 ** attempt * 2, 30))
    raise LibError(f"fetch failed after {retries} tries: {url} ({last})")
```

```python
"""多源记录去重合并入 catalog。stdlib only。"""
import argparse, json, sys
import lib

def record_score(r):
    s = 0
    if r.get("doi"):
        s += 3
    if r.get("abstract"):
        s += min(len(r["abstract"]) // 200, 5)
    if r.get("oa_url"):
        s += 1
    if r.get("venue"):
        s += 1
    if r.get("citation_count"):
        s += 1
    return s

def merge_records(records):
    by_doi, by_title, groups = {}, {}, []
    for r in records:
        d, t = lib.norm_doi(r.get("doi", "")), lib.norm_title(r.get("title", ""))
        g = None
        if d and d in by_doi:
            g = by_doi[d]
        elif t and t in by_title:
            g = by_title[t]
        if g is None:
            g = []
            groups.append(g)
        g.append(r)
        if d:
            by_doi[d] = g
        if t:
            by_title[t] = g
    out, merge_log = [], []
    for recs in groups:
        ordered = sorted(recs, key=record_score, reverse=True)
        best = {k: ordered[0].get(k, "" if k != "year" else None) for k in lib.UNIFIED_FIELDS}
        for r in ordered[1:]:
            for f in lib.UNIFIED_FIELDS:
                if not best.get(f) and r.get(f):
                    best[f] = r[f]
        best["n_sources"] = len({r.get("source") for r in recs})
        out.append(best)
        if len(recs) > 1:
            merge_log.append({"title": best["title"], "n_merged": len(recs),
                              "sources": [r.get("source") for r in recs]})
    out.sort(key=lambda r: (-(r.get("citation_count") or 0), lib.norm_title(r.get("title", ""))))
    for i, r in enumerate(out, 1):
        r["paper_id"] = f"P{i:03d}"
    return out, merge_log

def update_catalog(run_dir):
    catalog_path = os.path.join(run_dir, "catalog.json") if (os := __import__("os")) else None
    catalog = lib.load_json(catalog_path) or []
    merged, merge_log = merge_records(lib.load_records(run_dir))
    idx_doi = {lib.norm_doi(r["doi"]): r for r in catalog if r.get("doi")}
    idx_title = {lib.norm_title(r["title"]): r for r in catalog if r.get("title")}
    n_new = n_updated = 0
    for m in merged:
        hit = idx_doi.get(lib.norm_doi(m.get("doi", ""))) or \
              idx_title.get(lib.norm_title(m.get("title", "")))
        if hit:
            for f in lib.UNIFIED_FIELDS:
                if f != "source" or True:
                    hit[f] = m.get(f, hit.get(f))
            hit["n_sources"] = m["n_sources"]
            n_updated += 1
        else:
            e = lib.default_entry(m)
            catalog.append(e)
            if e["doi"]:
                idx_doi[lib.norm_doi(e["doi"])] = e
            if e["title"]:
                idx_title[lib.norm_title(e["title"])] = e
            n_new += 1
    lib.save_json(catalog_path, catalog)
    lib.save_json(os.path.join(run_dir, "dedup_report.json"),
                  {"n_raw": len(lib.load_records(run_dir)), "n_merged_groups": len(merged),
                   "n_new": n_new, "n_updated": n_updated, "merge_log": merge_log,
                   "date": lib.today()})
    return {"n_new": n_new, "n_updated": n_updated}

def check_dup(run_dir):
    catalog = lib.load_json(os.path.join(run_dir, "catalog.json")) or []
    seen_d, seen_t, dups = {}, {}, []
    for r in catalog:
        d, t = lib.norm_doi(r.get("doi", "")), lib.norm_title(r.get("title", ""))
        if d:
            if d in seen_d:
                dups.append(("doi", seen_d[d], r["paper_id"], d))
            seen_d[d] = r["paper_id"]
        if t:
            if t in seen_t:
                dups.append(("title", seen_t[t], r["paper_id"], r.get("title", "")[:60]))
            seen_t[t] = r["paper_id"]
    for kind, a, b, key in dups:
        print(f"DUP[{kind}] {a} == {b} ({key})")
    return 0 if not dups else 1

def main(argv=None):
    ap = argparse.ArgumentParser(description="dedup & merge raw records into catalog")
    ap.add_argument("--run", required=True)
    ap.add_argument("--check", action="store_true")
    a = ap.parse_args(argv)
    if a.check:
        return check_dup(a.run)
    stats = update_catalog(a.run)
    print(f"[dedup] new={stats['n_new']} updated={stats['n_updated']}")
    return 0

if __name__ == "__main__":
    sys.exit(main())
```

```python
BRIEF_FIELDS = ["项目描述", "所属领域", "主要目标", "调研深度", "时间范围", "资源约束"]
NOTE_SECTIONS = ["任务定义", "方法", "数据集", "基线", "指标", "结果", "局限", "证据引文"]
NOVELTY_BUCKETS = ("已覆盖", "可能差异待核验", "证据不支持")

def violation(rule, detail):
    return {"rule": rule, "detail": detail}

# ---------- 状态 ----------
def load_state(run):
    st = lib.load_json(os.path.join(run, "state.json"))
    if not st:
        sys.exit(f"gate: 找不到 {run}/state.json，先运行 gate.py new")
    return st

def save_state(run, st):
    st["updated"] = time.strftime("%Y-%m-%dT%H:%M:%S")
    lib.save_json(os.path.join(run, "state.json"), st)

# ---------- 解析 ----------
def parse_tree(run):
    p = os.path.join(run, "question_tree.md")
    if not os.path.exists(p):
        return []
    out = []
    for line in open(p, encoding="utf-8"):
        m = re.match(r"^\s*[-*]?\s*(Q\d+(?:\.\d+)*)\s+(.*)$", line.strip())
        if m:
            out.append((m.group(1), m.group(1).count(".") + 1, m.group(2)))
    return out

def parse_brief(run):
    p = os.path.join(run, "project_brief.md")
    if not os.path.exists(p):
        return {}
    fields = {}
    for line in open(p, encoding="utf-8"):
        m = re.match(r"^\s*[-*]?\s*([^：：]{2,8})[：:]\s*(.+)$", line.strip())
        if m and m.group(1) in BRIEF_FIELDS:
            fields[m.group(1)] = m.group(2).strip()
    return fields

# ---------- G1 ----------
def check_p1(run, st):
    v = []
    tier = lib.TIERS[st["tier"]]
    tree = parse_tree(run)
    if not tree:
        return [violation("G1.1", "question_tree.md 不存在或无 Q 节点")]
    core = [t for t in tree if t[1] == 1]
    if len(core) < tier["core_questions"]:
        v.append(violation("G1.1", f"一级核心问题 {len(core)} < 档位下限 {tier['core_questions']}"))
    for qid, depth, text in core:
        if "理由" not in text:
            v.append(violation("G1.2", f"{qid} 缺少'理由：'"))
    brief = parse_brief(run)
    for f in BRIEF_FIELDS:
        if not brief.get(f):
            v.append(violation("G1.3", f"project_brief.md 字段缺失或为空：{f}"))
    return v

# ---------- G2 ----------
def check_p2(run, st):
    v = []
    proto = lib.load_json(os.path.join(run, "search_protocol.json")) or {}
    tree = parse_tree(run)
    core = {qid for qid, depth, _ in tree if depth == 1}
    if not core:
        return [violation("G2.1", "问题树无一级节点（G1 未过？）")]
    queries = proto.get("queries") or []
    for q in sorted(core):
        for path in lib.PATHS:
            if not [x for x in queries if x.get("question_id") == q and x.get("path") == path]:
                v.append(violation("G2.1", f"{q} × {path} 无检索查询"))
    syn = proto.get("synonyms") or {}
    for q in sorted(core):
        if len([s for s in (syn.get(q) or []) if str(s).strip()]) < 3:
            v.append(violation("G2.2", f"{q} 同义词 <3"))
    for x in queries:
        if not x.get("date"):
            v.append(violation("G2.3", f"查询 {x.get('qid', '?')} 缺 date"))
    return v

# ---------- 执行 ----------
def run_check(run, phase):
    st = load_state(run)
    checker = CHECKERS.get(phase)
    if not checker:
        sys.exit(f"gate: 未知检查 {phase}")
    vs = checker(run, st)
    waived = {w["rule"] for w in st.get("waivers", [])}
    for x in vs:
        x["waived"] = x["rule"] in waived
    passed = all(x["waived"] for x in vs)
    st.setdefault("gate_results", {})[phase] = {
        "ts": time.strftime("%Y-%m-%dT%H:%M:%S"), "passed": passed,
        "n_violations": len([x for x in vs if not x["waived"]])}
    save_state(run, st)
    return vs, passed

def cmd_new(a):
    os.makedirs(a.run, exist_ok=True)
    save_state(a.run, {"project": a.project, "tier": a.tier, "phase": "P1",
                       "created": time.strftime("%Y-%m-%dT%H:%M:%S"),
                       "gate_results": {}, "waivers": [], "decisions": []})
    print(f"[gate] new run: {a.run} tier={a.tier}")

def cmd_check(a):
    vs, passed = run_check(a.run, a.phase)
    for x in vs:
        print(f"[{'WAIVED' if x['waived'] else 'FAIL'}] {x['rule']}: {x['detail']}")
    print(f"[gate] check {a.phase}: {'PASS' if passed else 'FAIL'}")
    return 0 if passed else 1

def cmd_set_phase(a):
    st = load_state(a.run)
    if a.phase not in PHASES:
        sys.exit(f"gate: 阶段必须是 {'/'.join(PHASES)}")
    cur, tgt = PHASES.index(st["phase"]), PHASES.index(a.phase)
    if tgt != cur + 1:
        sys.exit(f"gate: 只能顺序推进一格（当前 {st['phase']} → 请求 {a.phase}）")
    prev = PHASES[cur]
    vs, passed = run_check(a.run, prev)
    if not passed:
        for x in vs:
            if not x["waived"]:
                print(f"[FAIL] {x['rule']}: {x['detail']}")
        sys.exit(f"gate: {prev} 未通过，不能进入 {a.phase}")
    st["phase"] = a.phase
    st.setdefault("decisions", []).append(
        {"ts": time.strftime("%Y-%m-%dT%H:%M:%S"), "note": f"进入 {a.phase}"})
    save_state(a.run, st)
    print(f"[gate] phase -> {a.phase}")
    return 0

def cmd_waive(a):
    st = load_state(a.run)
    st.setdefault("waivers", []).append(
        {"rule": a.rule, "reason": a.reason, "date": lib.today()})
    save_state(a.run, st)
    print(f"[gate] waived {a.rule}: {a.reason}")

def cmd_status(a):
    st = load_state(a.run)
    print(json.dumps({"project": st.get("project"), "tier": st.get("tier"),
                      "phase": st.get("phase"),
                      "gate_results": st.get("gate_results"),
                      "waivers": st.get("waivers")}, ensure_ascii=False, indent=1))
    return 0

def main(argv=None):
    ap = argparse.ArgumentParser(description="质量门执法")

    s = sub.add_parser("set-phase"); s.add_argument("--run", required=True)
    s.add_argument("phase"); s.set_defaults(fn=cmd_set_phase)
    s = sub.add_parser("check"); s.add_argument("--run", required=True)
    s.add_argument("phase"); s.set_defaults(fn=cmd_check)
    s = sub.add_parser("waive"); s.add_argument("--run", required=True)
    s.add_argument("rule"); s.add_argument("--reason", required=True)
    s.set_defaults(fn=cmd_waive)
    s = sub.add_parser("status"); s.add_argument("--run", required=True)
    s.set_defaults(fn=cmd_status)
    a = ap.parse_args(argv)
    return a.fn(a)

if __name__ == "__main__":
    sys.exit(main())
```

```python
def check_p3(run, st):
    v = []
    tier = lib.TIERS[st["tier"]]
    cat = lib.load_json(os.path.join(run, "catalog.json")) or []
    if not cat:
        return [violation("G3.1", "catalog.json 为空或不存在")]
    seen = {}
    for r in cat:
        for kind, key in (("doi", lib.norm_doi(r.get("doi", ""))),
                          ("title", lib.norm_title(r.get("title", "")))):
            if not key:
                continue
            if (kind, key) in seen:
                v.append(violation("G3.2",
                        f"重复[{kind}] {seen[(kind, key)]} == {r.get('paper_id')}"))
            seen[(kind, key)] = r.get("paper_id")
    inc = [r for r in cat if r.get("reading_status") == "纳入"]
    for r in inc:
        if not r.get("question_ids"):
            v.append(violation("G3.3", f"{r['paper_id']} 纳入但未挂问题节点"))
    if len(inc) < tier["min_included"]:
        v.append(violation("G3.4", f"纳入文献 {len(inc)} < 档位下限 {tier['min_included']}"))
    n_rev = sum(1 for r in inc if r.get("paper_type") == "review")
    if n_rev < tier["min_reviews"]:
        v.append(violation("G3.4", f"综述 {n_rev} < 档位下限 {tier['min_reviews']}"))
    for r in cat:
        rs = r.get("reading_status")
        if rs == "待筛":
            v.append(violation("G3.5", f"{r['paper_id']} 仍待筛"))
        elif rs == "纳入" and not r.get("include_reason"):
            v.append(violation("G3.5", f"{r['paper_id']} 纳入缺 include_reason"))
        elif rs == "排除" and not r.get("exclude_reason"):
            v.append(violation("G3.5", f"{r['paper_id']} 排除缺 exclude_reason"))
    log = {e.get("query") for e in (lib.load_json(os.path.join(run, "search_log.json")) or [])}
    proto = lib.load_json(os.path.join(run, "search_protocol.json")) or {}
    for x in proto.get("queries") or []:
        if x.get("status") == "已执行" and x.get("query") not in log:
            v.append(violation("G3.6", f"{x.get('qid')} 声称已执行但 search_log 无记录"))
```

```python
def token_overlap(a, b):
    """标题 token 重叠率（0-1），分母取较短方。"""
    ta = set(re.findall(r"[a-z0-9]+", str(a or "").lower()))
    tb = set(re.findall(r"[a-z0-9]+", str(b or "").lower()))
    if not ta or not tb:
        return 0.0
    return len(ta & tb) / min(len(ta), len(tb))
```

```python
"""台账 → 报告/表格渲染。stdlib only。"""
import argparse, os, sys, time
import lib

DEFAULT_TEMPLATE = os.path.join(os.path.dirname(os.path.abspath(__file__)),
                                "..", "templates", "zh", "report.md")

def md_escape(s):
    return str(s or "").replace("|", "\\|").replace("\n", " ")

def table_catalog(run, include_only=True):
    cat = lib.load_json(os.path.join(run, "catalog.json")) or []
    rows = [r for r in cat if not include_only or r.get("reading_status") == "纳入"]
    rows.sort(key=lambda r: -(r.get("citation_count") or 0))
    lines = ["| paper_id | 标题 | 年 | 类型 | 引用 | 问题 | access |",
             "|---|---|---|---|---|---|---|"]
    for r in rows:
        lines.append(f"| {r.get('paper_id')} | {md_escape(r.get('title'))[:90]} | "
                     f"{r.get('year') or ''} | {r.get('paper_type','')} | "
                     f"{r.get('citation_count',0)} | "
                     f"{','.join(r.get('question_ids') or [])} | {r.get('access_status','')} |")
    return "\n".join(lines)

def table_gaps(run):
    gaps = lib.load_json(os.path.join(run, "gaps.json")) or []
    lines = ["| gap_id | 缺口 | 问题 | 类型 | 支撑 | 反证 | status |",
             "|---|---|---|---|---|---|---|"]
    for g in gaps:
        lines.append(f"| {g.get('gap_id')} | {md_escape(g.get('gap_statement'))[:80]} | "
                     f"{g.get('question_id','')} | {g.get('gap_type','')} | "
                     f"{','.join(g.get('supporting_evidence') or [])} | "
                     f"{','.join(g.get('counter_evidence') or [])} | {g.get('status','')} |")
    return "\n".join(lines)

def table_novelty(run):
    na = lib.load_json(os.path.join(run, "novelty_audit.json")) or {}
    lines = ["| direction | 主张 | bucket | 证据 | 先例 |", "|---|---|---|---|---|"]
    for d in na.get("directions") or []:
        for sc in d.get("subclaims") or []:
            lines.append(f"| {d.get('direction_id')} | {md_escape(sc.get('claim'))[:70]} | "
                         f"{sc.get('bucket','')} | {','.join(sc.get('evidence_ids') or [])} | "
                         f"{','.join(sc.get('prior_works') or [])} |")
    return "\n".join(lines)

def table_search_stats(run):
    log = lib.load_json(os.path.join(run, "search_log.json")) or []
    by_src, by_path = {}, {}
    for e in log:
        by_src[e.get("source", "?")] = by_src.get(e.get("source", "?"), 0) + 1
        by_path[e.get("path", "?")] = by_path.get(e.get("path", "?"), 0) + 1
    src = "、".join(f"{k}×{v}" for k, v in sorted(by_src.items()))
    path = "、".join(f"{k}×{v}" for k, v in sorted(by_path.items()))
    return f"- 检索源：{src}\n- 检索线：{path}\n- 查询总数：{len(log)}"

def waiver_table(run):
    st = lib.load_json(os.path.join(run, "state.json")) or {}
    ws = st.get("waivers") or []
    if not ws:
        return "- 豁免：无"
    lines = ["| 规则 | 理由 | 日期 |", "|---|---|---|"]
    for w in ws:
        lines.append(f"| {w.get('rule')} | {md_escape(w.get('reason'))[:90]} | {w.get('date','')} |")
    return "\n".join(lines)

def build_report(run, template=None, out=None):
    st = lib.load_json(os.path.join(run, "state.json")) or {}
    tpl = open(template or DEFAULT_TEMPLATE, encoding="utf-8").read()
    evid = lib.load_json(os.path.join(run, "evidence.json")) or []
    text = (tpl
            .replace("{{project_name}}", md_escape(st.get("project", "")))
            .replace("{{generated_at}}", lib.today())
            .replace("{{search_stats}}", table_search_stats(run))
            .replace("{{catalog_table}}", table_catalog(run))
            .replace("{{gap_table}}", table_gaps(run))
            .replace("{{novelty_table}}", table_novelty(run))
            .replace("{{waiver_table}}", waiver_table(run))
            .replace("{{evidence_count}}", str(len(evid))))
    out = out or os.path.join(run, "report.md")
    open(out, "w", encoding="utf-8").write(text)
    print(f"[render] {out}")
    return out

def main(argv=None):
    ap = argparse.ArgumentParser(description="台账渲染")
    sub = ap.add_subparsers(dest="cmd", required=True)
    s = sub.add_parser("report"); s.add_argument("--run", required=True)
    s.add_argument("--template"); s.add_argument("--out")
    s = sub.add_parser("table"); s.add_argument("--run", required=True)
    s.add_argument("--what", required=True,
                   choices=["catalog", "gaps", "search", "novelty"])
    a = ap.parse_args(argv)
    if a.cmd == "report":
        build_report(a.run, a.template, a.out)
        return 0
    fn = {"catalog": table_catalog, "gaps": table_gaps,
          "search": table_search_stats, "novelty": table_novelty}[a.what]
    print(fn(a.run))
    return 0

if __name__ == "__main__":
    sys.exit(main())
```

```markdown
# 项目背调（Project Due Diligence）

## 原则（区别于文献总结的四个机制）

1. 问题树是骨架：文献必须挂问题节点，结论必须挂问题；检索停止标准在树上。
2. 反证是结构性检索线：不主动找"已经有人做过了"，创新性结论不可信。
3. 三台账（catalog/evidence/gaps）+ 极性是知识形态；叙述只是其视图。
4. 交付决策件：缺口台账、创新审计、可行性风险、路线图缺一不可。

**禁止**：把"读了 N 篇"当完成条件；把作者自述局限当领域缺口；把单一来源当已证实；
把检索不到先例当作不存在先例（必须附检索范围声明）。
```
